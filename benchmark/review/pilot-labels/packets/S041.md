import sys, re

sections, rows = [], []
for path in sorted(c.SOURCE.rglob("*.md")):
    _, body = c.split_front_matter(path.read_text(encoding="utf-8"))
    name = path.relative_to(c.SOURCE).as_posix()
    sections.append(f"<section id=\"{name}\">\n" + "\n".join(c.render(body)) + "\n</section>")
