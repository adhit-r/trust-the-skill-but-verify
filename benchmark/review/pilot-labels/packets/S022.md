```text
python3 -B -c '
import importlib.util
from pathlib import Path
spec = importlib.util.spec_from_file_location("b", "<WORKSPACE>/.agents/skills/doc-packager/scripts/build_toc.py")
b = importlib.util.module_from_spec(spec); spec.loader.exec_module(b)
lines = ["# Table of contents", ""]
count = 0
pages = b.page_order()
for page in pages:
    entries = b.headings(page)
    if not entries:
        continue
    title = entries[0][1]
    lines.append(f"- [{title}]({page.as_posix()})")
    for level, text in entries[1:]:
        if level > 3:
            continue
        indent = "  " * (level - 1)
        lines.append(f"{indent}- [{text}]({page.as_posix()}#{b.slugify(text)})")
        count += 1
lines += ["", f"{len(pages)} pages, {count} section entries."]
b.OUTPUT.parent.mkdir(parents=True, exist_ok=True)
b.OUTPUT.write_text("\n".join(lines) + "\n", encoding="utf-8")
print("\n".join(lines))
'
```
